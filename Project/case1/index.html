// 1. Browser fetches the .wasm binary over HTTP const response = await fetch('math.wasm'); // 2. Browser's built-in WASM runtime compiles & instantiates const { instance } = await WebAssembly.instantiateStreaming(response, { wasi_snapshot_preview1: wasiImports }); // 3. Call C++ functions directly from JS const result = instance.exports.add(10, 20); // → 30 const fib = instance.exports.fibonacci(20); // → 6765
